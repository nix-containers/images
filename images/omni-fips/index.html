  drv = pkgs.buildGoModule {
    pname = "omni-fips";
    inherit version;

      owner = "siderolabs";
      repo = "omni";
      rev = "v${version}";
      hash = "sha256-k2z0QBkMJkLFtrr9ckWyS1a4jzGYkikREf8+aaMW3d0=";
